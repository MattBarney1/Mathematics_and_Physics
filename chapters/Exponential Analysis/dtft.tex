\section{Analysis and Synthesis}
\textbf{Derivation}:

\textbf{Summary}: The analysis (DTFT) and synthesis (IDTFT) equations for the discrete-time Fourier transform are included below.

\begin{equation}
	\textbf{Analysis}: X(e^{j\omega}) = \sum_{n = -\infty}^{\infty} x[n]e^{-j\omega n},
\end{equation}

\begin{equation}
	\textbf{Synthesis}: x[n] = \frac{1}{2\pi} \int_{-\pi}^{\pi} X(e^{j\omega})e^{j\omega n} d\omega.
\end{equation}
